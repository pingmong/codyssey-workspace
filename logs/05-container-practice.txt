## ubuntu 컨테이너 유지 (-d + sleep) 후 exec 로 진입
$ docker run -d --name ubu-keep ubuntu sleep 600
b61a55cf427d567520842e7bda8c443fee5c3337b57f9e54f5fcd7be59234fd5

$ docker exec ubu-keep bash -c 'echo "[exec] whoami=$(whoami)"; echo hello-from-exec > /tmp/note.txt; cat /tmp/note.txt'
[exec] whoami=root
hello-from-exec

$ docker exec ubu-keep ls -l /tmp
total 4
-rw-r--r-- 1 root root 16 Nov  2 05:30 note.txt
